#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <sys/stat.h>
#include <unistd.h>

#define SHM_NAME "/game_engine_buffer"
#define BUFFER_SIZE 4096

typedef struct {
    int frame_id;
    float delta_time;
    char payload[256];
} EngineState;

int init_shared_memory(void) {
    int shm_fd = shm_open(SHM_NAME, O_CREAT | O_RDWR, 0666);
    if (shm_fd == -1) {
        perror("shm_open failed");
        return -1;
    }

    if (ftruncate(shm_fd, BUFFER_SIZE) == -1) {
        perror("ftruncate failed");
        close(shm_fd);
        return -1;
    }

    EngineState *state = (EngineState *)mmap(
        NULL, BUFFER_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, shm_fd, 0
    );

    if (state == MAP_FAILED) {
        perror("mmap failed");
        close(shm_fd);
        return -1;
    }

    state->frame_id = 0;
    state->delta_time = 0.016f;
    strncpy(state->payload, "ENGINE_INITIALIZED", sizeof(state->payload));

    munmap(state, BUFFER_SIZE);
    close(shm_fd);
    return 0;
}
